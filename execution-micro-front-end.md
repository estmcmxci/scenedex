### Step 1: Define the Complete Release Data Model
**Deliverable:** `lib/types.ts`

```typescript
// The complete Release type (all states)
interface Release {
  id: string;                  // PDA-XXX (generated by system)
  title: string;
  description: string;
  media: string;               // ipfs://CID
  createdBy: string;           // 0xAddress
  zoraNFT?: string;            // base:0xContract/TokenID (set after L2 mint)
  metadataURI?: string;        // ipfs://CID of metadata (set after IPFS upload)
  tokenId?: string;            // From Zora mint (set after L2)
  status: 'pending' | 'approved' | 'published';
  createdAt: number;           // timestamp
  approvedAt?: number;         // timestamp when curator approved
  rejectionReason?: string;    // if rejected
}

// Input type for form submission (only user-provided fields)
type ReleaseSubmissionInput = Pick<Release, 'title' | 'description' | 'media'>;

// Curator approval action
interface CuratorAction {
  requestId: string;           // Release.id
  action: 'approve' | 'reject';
  reason?: string;             // if reject
}

// Pending request (what curator sees)
interface PendingRequest extends Release {
  status: 'pending';
}

// Published release (what catalog displays)
interface PublishedRelease extends Release {
  status: 'published';
  zoraNFT: string;             // must be set
  metadataURI: string;         // must be set
  tokenId: string;             // must be set
  approvedAt: number;          // must be set
}
```

**Why:** Every downstream piece now knows the exact shape of data. No guessing.

### Step 2: Define Validation Schema (Zod)
**Deliverable:** `lib/validation.ts`

```typescript
import { z } from 'zod';

// Validation for form input
export const ReleaseSubmissionSchema = z.object({
  title: z.string()
    .min(1, 'Title required')
    .max(200, 'Title max 200 chars'),
  description: z.string()
    .min(10, 'Description min 10 chars')
    .max(2000, 'Description max 2000 chars'),
  media: z.string()
    .regex(/^ipfs:\/\//, 'Must be IPFS CID (ipfs://...)')
    .max(100, 'Invalid IPFS CID'),
});

export type ReleaseSubmissionInput = z.infer<typeof ReleaseSubmissionSchema>;

// Full Release validation (used for storage/retrieval)
export const ReleaseSchema = z.object({
  id: z.string(),
  title: z.string(),
  description: z.string(),
  media: z.string(),
  createdBy: z.string().regex(/^0x/, 'Invalid Ethereum address'),
  zoraNFT: z.string().optional(),
  metadataURI: z.string().optional(),
  tokenId: z.string().optional(),
  status: z.enum(['pending', 'approved', 'published']),
  createdAt: z.number(),
  approvedAt: z.number().optional(),
  rejectionReason: z.string().optional(),
});

export type Release = z.infer<typeof ReleaseSchema>;
```

**Why:** Validation is colocated with types. Form, storage, API endpoints all use same schema.

### Step 4: Build Zustand Store (State Management)
**Deliverable:** `lib/store.ts`

```typescript
import { create } from 'zustand';
import { Release, CuratorAction } from '@/lib/types';
import { MOCK_RELEASES, MOCK_PENDING_REQUESTS } from '@/lib/mocks/fixtures';

interface CatalogueStore {
  // Submission flow
  submissions: Release[];
  submitRelease: (input: ReleaseSubmissionInput) => Promise<Release>;
  getSubmissionStatus: (requestId: string) => Release | null;

  // Curator board
  pendingRequests: Release[];
  fetchPendingRequests: () => Promise<void>;
  approveRelease: (requestId: string) => Promise<void>;
  rejectRelease: (requestId: string, reason: string) => Promise<void>;

  // Catalog display
  allReleases: Release[];
  fetchAllReleases: () => Promise<void>;
  getReleaseBySubname: (subname: string) => Promise<Release | null>;
}

export const useCatalogueStore = create<CatalogueStore>((set, get) => ({
  submissions: [],
  pendingRequests: MOCK_PENDING_REQUESTS,
  allReleases: MOCK_RELEASES.filter(r => r.status === 'published'),

  submitRelease: async (input: ReleaseSubmissionInput) => {
    const newRelease: Release = {
      id: `PDA-${String(Math.floor(Math.random() * 1000)).padStart(3, '0')}`,
      ...input,
      createdBy: '0x742d35Cc6634C0532925a3b844Bc9e7595f42e8', // from wallet
      status: 'pending',
      createdAt: Date.now(),
    };
    
    // Call mock API
    const res = await fetch('/api/submit', {
      method: 'POST',
      body: JSON.stringify(input),
    });
    
    const created = await res.json();
    set(state => ({ submissions: [...state.submissions, created] }));
    return created;
  },

  fetchPendingRequests: async () => {
    const res = await fetch('/api/curator/requests');
    const data = await res.json();
    set({ pendingRequests: data.pending });
  },

  approveRelease: async (requestId: string) => {
    const res = await fetch(`/api/curator/approve/${requestId}`, {
      method: 'POST',
    });
    
    if (res.ok) {
      set(state => ({
        pendingRequests: state.pendingRequests.filter(r => r.id !== requestId),
        allReleases: [
          ...state.allReleases,
          { ...state.pendingRequests.find(r => r.id === requestId)!, status: 'published' }
        ]
      }));
    }
  },

  fetchAllReleases: async () => {
    const res = await fetch('/api/releases');
    const data = await res.json();
    set({ allReleases: data.releases });
  },

  getReleaseBySubname: async (subname: string) => {
    // Implements fallback logic
    try {
      const res = await fetch(`/api/release/${subname}`);
      if (res.ok) return await res.json();
    } catch {
      // Fallback to ENS
      try {
        const res = await fetch(`/api/fallback/ens/${subname}`);
        if (res.ok) return await res.json();
      } catch {
        // Fallback to Zora
        const res = await fetch(`/api/fallback/zora/${subname}`);
        if (res.ok) return await res.json();
      }
    }
    return null;
  },
}));
```

**Why:** All state mutations go through the store. Components just dispatch actions. Easy to trace.

### Step 5: Set Up Mock Service Worker (MSW)
**Deliverable:** `lib/mocks/handlers.ts`

```typescript
import { http, HttpResponse } from 'msw';
import { Release, ReleaseSubmissionInput } from '@/lib/types';
import { MOCK_RELEASES, MOCK_PENDING_REQUESTS } from './fixtures';

let submissions: Release[] = [];
let pending: Release[] = [...MOCK_PENDING_REQUESTS];

export const handlers = [
  // Submit release
  http.post('/api/submit', async ({ request }) => {
    const input = (await request.json()) as ReleaseSubmissionInput;
    const newRelease: Release = {
      id: `PDA-${String(submissions.length + 1).padStart(3, '0')}`,
      ...input,
      createdBy: '0x742d35Cc6634C0532925a3b844Bc9e7595f42e8',
      status: 'pending',
      createdAt: Date.now(),
    };
    submissions.push(newRelease);
    pending.push(newRelease);
    return HttpResponse.json(newRelease, { status: 201 });
  }),

  // Get pending requests (for curator)
  http.get('/api/curator/requests', () => {
    return HttpResponse.json({ pending });
  }),

  // Approve release
  http.post('/api/curator/approve/:requestId', ({ params }) => {
    const { requestId } = params;
    const release = pending.find(r => r.id === requestId);
    if (release) {
      release.status = 'approved';
      release.approvedAt = Date.now();
      pending = pending.filter(r => r.id !== requestId);
      return HttpResponse.json({ approved: true, release });
    }
    return HttpResponse.json({ error: 'Not found' }, { status: 404 });
  }),

  // Get all published releases
  http.get('/api/releases', () => {
    const published = MOCK_RELEASES.filter(r => r.status === 'published');
    return HttpResponse.json({ releases: published });
  }),

  // Get single release (with fallback simulation)
  http.get('/api/release/:subname', ({ params }) => {
    const release = MOCK_RELEASES.find(r => r.id === params.subname);
    return release 
      ? HttpResponse.json(release)
      : HttpResponse.json({ error: 'Not found' }, { status: 404 });
  }),

  // Fallback endpoints
  http.get('/api/fallback/ens/:subname', ({ params }) => {
    return HttpResponse.json(MOCK_RELEASES[0]); // Simulate ENS resolver
  }),

  http.get('/api/fallback/zora/:subname', ({ params }) => {
    return HttpResponse.json(MOCK_RELEASES[0]); // Simulate Zora API
  }),
];
```

**Why:** MSW intercepts at network level. Phase 1 hits these. Phase 3 swaps endpoints, same flow.

### Step 6: Build MetadataForm Component

```typescript
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { ReleaseSubmissionSchema, ReleaseSubmissionInput } from '@/lib/validation';
import { useCatalogueStore } from '@/lib/store';
import { useState } from 'react';

export function MetadataForm() {
  const [submitted, setSubmitted] = useState(false);
  const [requestId, setRequestId] = useState<string | null>(null);
  const submitRelease = useCatalogueStore(s => s.submitRelease);

  const {
    register,
    handleSubmit,
    formState: { errors },
  } = useForm<ReleaseSubmissionInput>({
    resolver: zodResolver(ReleaseSubmissionSchema),
  });

  const onSubmit = async (data: ReleaseSubmissionInput) => {
    const release = await submitRelease(data);
    setRequestId(release.id);
    setSubmitted(true);
  };

  if (submitted) {
    return (
      <div className="p-8 bg-green-50 border border-green-200 rounded-lg">
        <h2 className="text-2xl font-bold text-green-900">Release Submitted!</h2>
        <p className="mt-2 text-green-700">Request ID: {requestId}</p>
        <p className="mt-4 text-green-600">Awaiting curator board approval...</p>
      </div>
    );
  }

  return (
    <form onSubmit={handleSubmit(onSubmit)} className="space-y-6 max-w-2xl">
      <div>
        <label className="block text-sm font-medium">Title</label>
        <input
          {...register('title')}
          className="mt-1 w-full px-4 py-2 border rounded-lg"
          placeholder="Release title"
        />
        {errors.title && <p className="text-red-600 text-sm">{errors.title.message}</p>}
      </div>

      <div>
        <label className="block text-sm font-medium">Description</label>
        <textarea
          {...register('description')}
          className="mt-1 w-full px-4 py-2 border rounded-lg"
          placeholder="Release description (min 10 chars)"
          rows={4}
        />
        {errors.description && <p className="text-red-600 text-sm">{errors.description.message}</p>}
      </div>

      <div>
        <label className="block text-sm font-medium">Media IPFS CID</label>
        <input
          {...register('media')}
          className="mt-1 w-full px-4 py-2 border rounded-lg"
          placeholder="ipfs://QmExample..."
        />
        {errors.media && <p className="text-red-600 text-sm">{errors.media.message}</p>}
      </div>

      <button
        type="submit"
        className="w-full bg-blue-600 text-white py-2 rounded-lg font-semibold hover:bg-blue-700"
      >
        Submit Release
      </button>
    </form>
  );
}
```

### Step 7: Build CuratorDashboard Component
**Deliverable:** `components/CuratorDashboard.tsx`

```typescript
'use client';

import { useEffect, useState } from 'react';
import { useCatalogueStore } from '@/lib/store';
import { Release } from '@/lib/types';

export function CuratorDashboard() {
  const { pendingRequests, fetchPendingRequests, approveRelease } = useCatalogueStore();
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchPendingRequests().finally(() => setLoading(false));
  }, []);

  if (loading) return <div>Loading pending requests...</div>;
  if (pendingRequests.length === 0) return <div>No pending requests</div>;

  return (
    <div className="space-y-4">
      <h1 className="text-3xl font-bold">Curator Board</h1>
      {pendingRequests.map(request => (
        <ApprovalCard key={request.id} release={request} />
      ))}
    </div>
  );
}

function ApprovalCard({ release }: { release: Release }) {
  const approveRelease = useCatalogueStore(s => s.approveRelease);
  const [approving, setApproving] = useState(false);

  const handleApprove = async () => {
    setApproving(true);
    await approveRelease(release.id);
    setApproving(false);
  };

  return (
    <div className="p-6 border rounded-lg bg-white">
      <h2 className="text-xl font-bold">{release.title}</h2>
      <p className="text-gray-600 mt-2">{release.description}</p>
      <p className="text-sm text-gray-500 mt-2">From: {release.createdBy}</p>
      <button
        onClick={handleApprove}
        disabled={approving}
        className="mt-4 px-4 py-2 bg-green-600 text-white rounded-lg hover:bg-green-700 disabled:opacity-50"
      >
        {approving ? 'Approving...' : 'Approve'}
      </button>
    </div>
  );
}
```

**Why:** Component reads Release[] from store. Maps over typed releases. Each card is a PresentationalComponent.

### Step 9: Build ReleaseDetail Page
**Deliverable:** `app/release/[subname]/page.tsx`

```typescript
'use client';

import { useEffect, useState } from 'react';
import { useCatalogueStore } from '@/lib/store';
import { Release } from '@/lib/types';

export default function ReleaseDetailPage({ params }: { params: { subname: string } }) {
  const { getReleaseBySubname } = useCatalogueStore();
  const [release, setRelease] = useState<Release | null>(null);
  const [loading, setLoading] = useState(true);
  const [fallbackUsed, setFallbackUsed] = useState(false);

  useEffect(() => {
    getReleaseBySubname(params.subname)
      .then(r => {
        if (r) {
          setRelease(r);
          if (params.subname !== 'PDA-001') setFallbackUsed(true);
        }
      })
      .finally(() => setLoading(false));
  }, [params.subname]);

  if (loading) return <div>Loading...</div>;
  if (!release) return <div>Release not found</div>;

  return (
    <div className="max-w-2xl mx-auto p-8">
      <h1 className="text-4xl font-bold">{release.title}</h1>
      <p className="text-xl text-gray-600 mt-4">{release.description}</p>

      <div className="mt-8 space-y-4 bg-gray-50 p-6 rounded-lg">
        <div>
          <p className="text-sm text-gray-600">Catalog ID</p>
          <p className="text-lg font-mono font-bold">{release.id}</p>
        </div>
        <div>
          <p className="text-sm text-gray-600">Creator</p>
          <p className="text-lg font-mono">{release.createdBy}</p>
        </div>
        {release.zoraNFT && (
          <div>
            <p className="text-sm text-gray-600">Zora NFT</p>
            <p className="text-lg font-mono">{release.zoraNFT}</p>
          </div>
        )}
        {release.metadataURI && (
          <div>
            <p className="text-sm text-gray-600">Metadata URI</p>
            <a href={`https://ipfs.io/ipfs/${release.metadataURI.replace('ipfs://', '')}`} 
               target="_blank" 
               className="text-blue-600 hover:underline">
              View on IPFS
            </a>
          </div>
        )}
      </div>

      {fallbackUsed && (
        <div className="mt-8 p-4 bg-yellow-50 border border-yellow-200 rounded-lg text-sm text-yellow-800">
          ℹ️ Data retrieved from fallback source (The Graph unavailable)
        </div>
      )}
    </div>
  );
}
```
